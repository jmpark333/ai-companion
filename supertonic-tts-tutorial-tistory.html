🎤 Supertonic TTS 브라우저 구현 가이드

Supertonic TTS는 브라우저에서 직접 실행되는 고품질 텍스트 음성 변환(Text-to-Speech) 라이브러리입니다. 이 가이드는 온디바이스(on-device) TTS 구현의 전체 과정을 다루며, 초보자부터 중급자까지 단계별로 따라할 수 있는 실용적인 튜토리얼을 제공합니다.

🚀 Supertonic TTS란?

Supertonic TTS는 다음 특징을 갖는 혁신적인 음성 합성 라이브러리입니다:

  • 온디바이스 처리: 서버 없이 브라우저에서 직접 실행
  • 고속 처리: 기존 TTS보다 최대 10배 빠른 성능
  • 최상급 음질: 실제 사람과 구분되지 않는 자연스러운 음성
  • 프라이버시 보호: 텍스트가 외부로 전송되지 않음
  • 크로스플랫폼: 모든 주요 브라우저에서 동작
  • WebGPU 지원: 최신 GPU 가속 기술 지원

📋 사전 준비 사항

시스템 요구사항

  • 브라우저: Chrome 88+, Firefox 84+, Safari 14+, Edge 88+
  • 메모리: 최소 4GB RAM (모델 로딩을 위해)
  • 저장공간: 약 400MB (모델 파일 포함)
  • 네트워크: 초기 모델 다운로드 시 필요

2단계: 음성 스타일 파일 설정

💡 음성 스타일 파일 설명: Supertonic TTS는 style_ttl과 style_dp 두 개의 텐서를 사용하여 다양한 음성 특성을 표현합니다. 각 스타일 파일은 특정 음성의 성격과 톤을 정의합니다. 
/**
 * 올바른 음성 스타일 텐서 로딩 함수
 */
async function loadVoiceStyle(voiceStylePaths) {
    try {
        // 첫 번째 스타일 파일에서 차원 정보 추출
        const firstStyleResponse = await fetch(voiceStylePaths[0]);
        const firstStyle = await firstStyleResponse.json();
        
        // 텐서 차원 확인
        const ttlDims = firstStyle.style_ttl.dims;    // 예: [1, 10, 768]
        const dpDims = firstStyle.style_dp.dims;      // 예: [1, 5, 256]
        
        if (ttlDims[0] !== 1 || dpDims[0] !== 1) {
            throw new Error('Supertonic TTS는 단일 화자만 지원합니다.');
        }
        
        // 텐서 데이터 평탄화
        const ttlFlat = new Float32Array(firstStyle.style_ttl.data.flat(Infinity));
        const dpFlat = new Float32Array(firstStyle.style_dp.data.flat(Infinity));
        
        // ONNX Runtime 텐서 생성
        const ttlTensor = new ort.Tensor('float32', ttlFlat, ttlDims);
        const dpTensor = new ort.Tensor('float32', dpFlat, dpDims);
        
        return new Style(ttlTensor, dpTensor);
        
    } catch (error) {
        console.error('음성 스타일 로딩 실패:', error);
        throw error;
    }
}

3단계: 핵심 TTS 클래스 구현

/**
 * Supertonic TTS 메인 클래스
 * Hugging Face Space 예제를 기반으로 한 브라우저 구현
 */
class SupertonicTTS {
    constructor() {
        this.textToSpeech = null;
        this.cfgs = null;
        this.currentStyle = null;
        this.modelLoaded = false;
        this.isLoading = false;
        this.executionProvider = 'wasm'; // webgpu 또는 wasm
        
        // 기본 설정
        this.config = {
            totalSteps: 5,      // 추론 스텝 수 (1-10 권장)
            speed: 1.05,        // 음성 속도 (0.5-2.0)
            voiceStyle: 'F1',   // 기본 음성 스타일
            baseUrl: '/assets'  // 모델 파일 기준 경로
        };
    }

    /**
     * TTS 엔진 초기화
     */
    async initialize() {
        if (this.modelLoaded || this.isLoading) {
            console.log('모델이 이미 로드되었거나 로드 중입니다.');
            return this.modelLoaded;
        }

        this.isLoading = true;
        console.log('Supertonic TTS 모델 로딩 시작...');

        try {
            // Web Audio API 확인
            if (!window.AudioContext) {
                throw new Error('이 브라우저는 Web Audio API를 지원하지 않습니다.');
            }
            
            // ONNX Runtime Web 확인
            if (!window.ort) {
                throw new Error('ONNX Runtime Web이 로드되지 않았습니다. CDN에서 ort.min.js를 로드하세요.');
            }

            // WebGPU 우선 시도
            try {
                const result = await this.loadModels(this.config.baseUrl, {
                    executionProviders: ['webgpu'],
                    graphOptimizationLevel: 'all'
                });
                
                this.textToSpeech = result.textToSpeech;
                this.cfgs = result.cfgs;
                this.executionProvider = 'webgpu';
                
                console.log('✅ WebGPU 백엔드로 모델 로드 완료');
            } catch (webgpuError) {
                console.log('WebGPU를 지원하지 않음, WebAssembly로 폴백');
                
                const result = await this.loadModels(this.config.baseUrl, {
                    executionProviders: ['wasm'],
                    graphOptimizationLevel: 'all'
                });
                
                this.textToSpeech = result.textToSpeech;
                this.cfgs = result.cfgs;
                this.executionProvider = 'wasm';
                
                console.log('✅ WebAssembly 백엔드로 모델 로드 완료');
            }
            
            // 음성 스타일 로드
            const stylePath = `/assets/voice_styles/${this.config.voiceStyle}.json`;
            this.currentStyle = await loadVoiceStyle([stylePath]);
            
            this.modelLoaded = true;
            this.isLoading = false;
            
            console.log(`✅ Supertonic TTS 모델 로딩 완료 (${this.executionProvider.toUpperCase()} 백엔드)`);
            return true;
            
        } catch (error) {
            this.isLoading = false;
            console.error('❌ TTS 모델 로딩 실패:', error);
            throw new Error(`TTS 모델 로딩 실패: ${error.message}`);
        }
    }

    /**
     * 음성 생성 메인 함수
     */
    async speak(text, options = {}) {
        if (!this.modelLoaded) {
            await this.initialize();
        }

        const {
            progressCallback = () => {},
            totalSteps = this.config.totalSteps,
            speed = this.config.speed
        } = options;

        try {
            const { wav, duration } = await this.textToSpeech.call(
                text, 
                this.currentStyle, 
                totalSteps,
                speed,
                0.3,
                progressCallback
            );
            
            // WAV 파일 생성 및 재생
            const wavLen = Math.floor(this.textToSpeech.sampleRate * duration[0]);
            const wavOut = wav.slice(0, wavLen);
            
            return {
                audioBuffer: wavOut,
                duration: duration[0],
                sampleRate: this.textToSpeech.sampleRate
            };
            
        } catch (error) {
            console.error('음성 생성 실패:', error);
            throw error;
        }
    }
}

}

⚠️ 흔한 문제점과 해결법

모델 로딩 오류

⚠️ 증상: "ONNX Runtime Web not found" 또는 모델 파일 로딩 실패
원인 및 해결:
  • ONNX Runtime 미로드: CDN 스크립트 추가
  • CORS 정책 오류: 로컬에서 file:/// 프로토콜 사용 시 발생
    💡 해결법: 로컬 서버를 실행하거나 HTTPS 도메인에서 테스트하세요.
  • 메모리 부족: 큰 모델 파일 로딩 시 발생
    💡 해결법: 브라우저 탭을 다시 시작하거나 더 많은 RAM이 있는 기기 사용

텐서 형태 오류

⚠️ 증상: "input 'style_dp' is missing in 'feeds'"
원인 및 해결: 
// ❌ 잘못된 방식
const style = {
    dp: [],           // 빈 배열
    ttl: []           // 빈 배열
};

// ✅ 올바른 방식
const style = {
    dp: ort.Tensor('float32', dpData, dpShape),   // ORT 텐서
    ttl: ort.Tensor('float32', ttlData, ttlShape) // ORT 텐서
};

오디오 재생 문제

⚠️ 증상: 음성이 생성되지만 재생되지 않음
해결 방법:
  • AudioContext 중단: 사용자 상호작용 후 resume() 호출 
    // 오디오 재생 전 AudioContext 활성화
if (this.audioContext && this.audioContext.state === 'suspended') {
    await this.audioContext.resume();
}
  • 데이터 형태 오류: Float32Array 형태로 변환 확인 
    // AudioBuffer 생성 시 올바른 데이터 형태 사용
const audioBuffer = this.audioContext.createBuffer(1, audioData.length, 44100);
const channelData = audioBuffer.getChannelData(0);

// Float32Array로 변환하여 복사
for (let i = 0; i < audioData.length; i++) {
    channelData[i] = Math.max(-1, Math.min(1, audioData[i]));
}

⚡ 성능 최적화 가이드

메모리 관리

💡 중요한 최적화 팁: Supertonic TTS는 메모리를 많이 사용하는 라이브러리입니다. 다음 최적화 방법을 적용하세요:
  1. 모델 캐싱: 한 번 초기화 후 인스턴스 재사용 
    // 전역 TTS 인스턴스 생성 (한 번만)
window.ttsEngine = new SupertonicTTS();

// 여러 번 재사용
async function speakMultipleTexts(texts) {
    if (!window.ttsEngine.modelLoaded) {
        await window.ttsEngine.initialize();
    }
    
    for (const text of texts) {
        await window.ttsEngine.speak(text);
    }
}
  2. 메모리 정리: 사용하지 않는 텐서 정리 
    // 큰 배열 사용 후 수동 정리
function processAudioData(audioData) {
    try {
        // 오디오 데이터 처리
        const result = processAudio(audioData);
        return result;
    } finally {
        // 명시적 메모리 정리
        audioData = null;
        if (global.gc) {
            global.gc(); // Node.js의 경우
        }
    }

  2. 텍스트 청킹: 긴 텍스트를 여러小块로 분할 
    // 긴 텍스트 자동 분할
function chunkText(text, maxLen = 300) {
    const sentences = text.split(/[.!?]+/);
    const chunks = [];
    let currentChunk = '';
    
    for (const sentence of sentences) {
        if (currentChunk.length + sentence.length > maxLen) {
            if (currentChunk) chunks.push(currentChunk);
            currentChunk = sentence;
        } else {
            currentChunk += (currentChunk ? '. ' : '') + sentence;
        }
    }
    
    if (currentChunk) chunks.push(currentChunk);
    return chunks;
}

💡 중요한 권장사항: Supertonic TTS를 프로덕션 환경에서 사용할 때 다음 사항들을 반드시 고려하세요:
  1. 에러 핸들링: Web Speech API를 폴백으로 사용 
    async function robustTextToSpeech(text) {
    try {
        // Supertonic TTS 시도
        return await supersonicTTS.speak(text);
    } catch (error) {
        console.warn('Supertonic TTS 실패, Web Speech API 폴백:', error);

        if ('speechSynthesis' in window) {
            return new Promise((resolve, reject) => {
                const utterance = new SpeechSynthesisUtterance(text);
                utterance.lang = 'ko-KR';
                utterance.rate = 0.9;
                
                utterance.onend = () => resolve(true);
                utterance.onerror = (e) => reject(e.error);
                
                speechSynthesis.speak(utterance);
            });
        }
        
        throw error;
    }
}

    // 기능 지원 확인
function checkBrowserSupport() {
    const support = {
        onnx: !!window.ort,
        webAudio: !!window.AudioContext,
        webSpeech: 'speechSynthesis' in window,
        webGPU: 'gpu' in navigator
    };
    
    if (!support.onnx || !support.webAudio) {
        console.warn('Supertonic TTS를 지원하지 않는 브라우저입니다.');
        showFallbackOption();
    }
    
    return support;
}

프로덕션 고려사항

  • 모델 최적화: 필요한 모델만 로드하여 메모리 사용량 최소화
  • CDN 활용: 모델 파일을 CDN에서 로드하여 다운로드 속도 향상
  • 로컬 스토리지: 모델을 로컬에 캐시하여 재사용
  • 모니터링: 성능 지표와 오류율 추적

흔한 오류 코드

오류 메시지 원인 해결 방법
input 'style_dp' is missing 스타일 텐서 형태 오류 ONNX 텐서 형태로 변환 확인
AudioContext was not allowed 사용자 상호작용 없음 사용자 클릭 후 AudioContext.resume()
Model file not found 모델 파일 경로 오류 파일 경로와 CORS 정책 확인
Out of memory 메모리 부족 브라우저 재시작 또는 탭 최소화
WebGPU not available WebGPU 지원 안됨 WebAssembly로 자동 폴백됨
